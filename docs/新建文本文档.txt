一、核心设计问题
1.1 库存计算逻辑存在严重缺陷
问题描述：
方案中说"套餐可售数量 = min(各底层 item 当日 availableStock / item.quantity)"，但酒店房型的库存是按每晚扣减的，跨越多天时计算逻辑错误。

举例：

text
套餐：豪华大床房2晚 + 景区门票2张
- 5月1日：房型库存=1，门票库存=100
- 5月2日：房型库存=10，门票库存=100

按方案公式：
  房型可用 = min(1, 10) = 1  ← 错误！应该看连续日期
  实际计算结果是1份，但5月2日的库存是够的
修复方案：

javascript
// 酒店套餐：需要检查连续日期的库存最小值
async function getHotelPackageStock(roomTypeId, checkIn, checkOut, quantity) {
  const dates = generateDateRange(checkIn, checkOut);
  let minStock = Infinity;
  
  for (const date of dates) {
    const stock = await RoomDatePrice.findOne({
      where: { roomPackageId: roomTypeId, date }
    });
    // 每晚都需要足够库存
    minStock = Math.min(minStock, stock.availableStock);
  }
  // 返回按需要数量折算后的可售份数
  return Math.floor(minStock / quantity);
}

// 最终套餐可售数量 = min(房型可售, 各门票可售)
1.2 主产品定义与业务规则矛盾
问题描述：
方案说"每个套餐有且仅有1个主产品，日期和库存由主产品驱动"，但酒店套餐主产品是房型，门票套餐主产品是门票。

矛盾点：

酒店套餐：用户选入住/离店日期，门票游玩日期=入住日期 ✅ 成立

门票套餐：用户选游玩日期，主产品是门票 ✅ 成立

但如果酒店套餐里包含2个不同使用日期的门票呢？比如入住当天用的门票 + 离店当天用的门票？

场景举例：

text
"周末放松套餐"：周五入住+周六离店，包含：
  - 周五晚上温泉门票（使用日期=入住当天）
  - 周六上午景区门票（使用日期=退房当天，即入住日期+1）
此时"门票游玩日期=入住日期"就不成立了。

修复方案：

sql
-- package_items 表新增字段
ALTER TABLE package_items ADD COLUMN `useDateOffset` INT DEFAULT 0 COMMENT '日期偏移天数，如门票在入住后N天使用';
javascript
// 计算使用日期
function getUseDate(item, checkIn, checkOut, playDate) {
  if (item.bizType === 'hotel') {
    return { startDate: checkIn, endDate: checkOut };
  }
  
  switch(item.useDateType) {
    case 1: // 跟随入住日期
      return { startDate: addDays(checkIn, item.useDateOffset || 0), endDate: null };
    case 2: // 跟随游玩日期
      return { startDate: playDate, endDate: null };
    case 3: // 固定日期
      return { startDate: item.useDateValue, endDate: null };
    case 4: // 购买后N天内有效
      return { startDate: null, endDate: addDays(new Date(), item.useDateValue) };
    default:
      return { startDate: checkIn, endDate: null };
  }
}

1.4 价格与库存的脱节问题
问题描述：
package_date_prices 存套餐价格，package_price_rules 存价格规则。但套餐价格与底层 items 价格之和没有关联校验。

风险：

text
套餐成本（底层items价格之和）= 房型699 + 门票120*2 = 939元
套餐售价设置 = 399元  ← 卖一套亏一套
修复方案：

sql
-- product_packages 表新增字段
ALTER TABLE product_packages ADD COLUMN `costPrice` DECIMAL(10,2) DEFAULT 0.00 COMMENT '成本价（自动计算）';
ALTER TABLE product_packages ADD COLUMN `profitMargin` DECIMAL(5,2) DEFAULT NULL COMMENT '毛利率（提醒用）';
javascript
// 创建/编辑套餐时自动校验
async function validatePackagePrice(packageId, items, salePrice) {
  // 计算各item原价之和
  let totalCost = 0;
  for (const item of items) {
    // 取 item.unitPrice 或查询底层业务表的最新价格
    totalCost += item.unitPrice * item.quantity;
  }
  
  if (salePrice < totalCost * 0.5) {
    console.warn(`警告：套餐${packageId}售价${salePrice}低于成本价50%`);
  }
  
  // 更新成本价
  await ProductPackage.update(
    { costPrice: totalCost },
    { where: { id: packageId } }
  );
}

2.2 核销后的库存同步问题
问题：
套餐核销后，package_order_details 的核销状态更新了，但底层业务表（如酒店入住状态）没有同步更新。

场景：
用户核销了酒店住宿，但酒店系统不知道这个订单已经确认入住。

修复方案：

javascript
// 核销时同步调用业务系统接口
async function verifyDetail(verifyCode) {
  await sequelize.transaction(async (t) => {
    const detail = await PackageOrderDetail.findOne({
      where: { verifyCode },
      lock: t.LOCK.UPDATE
    });
    
    await detail.update({ verifyStatus: 2, verifyTime: new Date() }, { transaction: t });
    
    // 调用业务系统核销接口
    if (detail.bizType === 'hotel') {
      await hotelService.checkin({
        orderNumber: detail.orderNumber,
        roomTypeId: detail.bizId,
        useDate: detail.useDate
      });
    } else if (detail.bizType === 'ticket') {
      await ticketService.useTicket({
        orderNumber: detail.orderNumber,
        ticketId: detail.bizId,
        verifyCode: detail.verifyCode
      });
    }
  });
}
2.3 订单取消时库存释放的并发问题
问题：
订单超时未支付，定时任务释放库存。如果用户在超时临界点恰好支付，会出现支付成功但订单已被取消的情况。

修复方案：

sql
-- orders 表新增版本号字段，用于乐观锁
ALTER TABLE orders ADD COLUMN `version` INT DEFAULT 0;
javascript
// 支付回调时检查版本
async function payCallback(orderNumber) {
  const result = await sequelize.transaction(async (t) => {
    const order = await Order.findOne({
      where: { orderNumber, orderStatus: 1 }, // 只有待支付才能支付
      lock: t.LOCK.UPDATE
    });
    
    if (!order) {
      throw new Error('订单不存在或已取消');
    }
    
    // 检查是否过期
    if (order.expireTime < new Date()) {
      throw new Error('订单已过期');
    }
    
    // 更新订单状态（同时更新版本号）
    const affected = await Order.update(
      { orderStatus: 2, version: order.version + 1 },
      { where: { orderNumber, version: order.version }, transaction: t }
    );
    
    if (affected[0] === 0) {
      throw new Error('订单状态已变更，请重试');
    }
  });
}

4.3 并发库存检查的边界情况
问题：
方案中的库存检查只校验了 availableStock >= needStock，但没有考虑 maxSaleNum 和底层库存的联动。

场景：

text
套餐 maxSaleNum=50，已卖45份
底层酒店房型5月1日库存=10份

用户A买3份 → 校验通过（48<50，底层够）
用户B买3份 → 校验通过（51>50？需要检查）

问题：两个请求同时到达，都会通过 maxSaleNum 校验
修复：

javascript
// 使用数据库行锁 + 原子操作
async function deductStockAtomic(packageId, buyNum, transaction) {
  // 原子性更新 soldStock，同时检查是否超限
  const [affected] = await sequelize.query(`
    UPDATE product_packages 
    SET soldStock = soldStock + :buyNum,
        status = IF(soldStock + :buyNum >= maxSaleNum AND maxSaleNum > 0, 3, status)
    WHERE id = :packageId 
      AND (maxSaleNum = 0 OR soldStock + :buyNum <= maxSaleNum)
  `, {
    replacements: { packageId, buyNum },
    transaction
  });
  
  if (affected === 0) {
    throw new Error('套餐已售罄');
  }
}

六、总结建议优先级
优先级	问题	建议	影响
P0	多天酒店库存计算错误	修复为逐日取最小值	会导致超卖
P0	订单表字段类型不匹配	用扩展表或改表结构	会导致数据异常
P1	事务范围过大	两阶段提交或拆分事务	高并发死锁
P1	combine类型日期计算	支持日期偏移	业务场景受限
P2	价格与成本脱节	增加成本校验	运营亏损
P2	缓存设计缺失	补充Redis缓存	性能问题
P3	核销后业务系统同步	调用业务系统接口	状态不一致
整体设计思路正确，但细节上需要优化。建议优先处理P0和P1的问题后再进入开发。
